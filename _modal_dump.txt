'use client';

import { useState, useEffect } from 'react';
import { supabase } from '../supabase';

export default function EditCoordinatorsModal({
  session,
  currentCoordinators,
  onClose,
  onSaved,
}: {
  session: any;
  currentCoordinators: any[];
  onClose: () => void;
  onSaved: () => void;
}) {
  const [facultyList, setFacultyList] = useState<any[]>([]);
  const [searchFaculty, setSearchFaculty] = useState('');
  const [selectedRole, setSelectedRole] = useState('National');
  const [coordinators, setCoordinators] = useState<any[]>([]);
  const [isSaving, setIsSaving] = useState(false);

  // Load all faculty and initialize current coordinators
  useEffect(() => {
    async function loadFaculty() {
      const { data } = await supabase.from('faculty').select('id, canonical_name, email').order('canonical_name');
      if (data) setFacultyList(data);
    }
    loadFaculty();
    // The currentCoordinators prop includes the faculty join, so we can use it directly
    setCoordinators(currentCoordinators);
  }, [currentCoordinators]);

  const filteredFaculty = facultyList.filter(f => 
    f.canonical_name.toLowerCase().includes(searchFaculty.toLowerCase()) ||
    f.email?.toLowerCase().includes(searchFaculty.toLowerCase())
  ).slice(0, 5);

  async function handleAddCoordinator(faculty: any) {
    if (coordinators.find(c => c.faculty_id === faculty.id)) {
      alert("This faculty is already a coordinator for this session.");
      return;
    }
    // Optimistically add to UI
    const newCoord = {
      session_id: session.id,
      faculty_id: faculty.id,
      role: selectedRole,
      faculty: faculty
    };
    setCoordinators([...coordinators, newCoord]);
    setSearchFaculty('');
  }

  async function handleRemoveCoordinator(faculty_id: string) {
    setCoordinators(coordinators.filter(c => c.faculty_id !== faculty_id));
  }

  async function handleSave() {
    setIsSaving(true);

    // 1. Delete all existing coordinators for this session
    await supabase.from('session_coordinators').delete().eq('session_id', session.id);

    // 2. Insert the new list of coordinators
    if (coordinators.length > 0) {
      const payload = coordinators.map(c => ({
        session_id: session.id,
        faculty_id: c.faculty.id || c.faculty_id, // Handle both fresh and existing
        role: c.role
      }));

      const { error } = await supabase.from('session_coordinators').insert(payload);
      
      if (error) {
        alert("Error saving coordinators: " + error.message);
        setIsSaving(false);
        return;
      }
    }

    setIsSaving(false);
    onSaved();
    onClose();
  }

  return (
    <div className="fixed inset-0 bg-black/50 flex items-center justify-center z-50 p-4">
      <div className="bg-white rounded-xl shadow-xl w-full max-w-lg overflow-hidden">
        <div className="flex justify-between items-center p-5 border-b bg-gray-50">
          <div>
            <h2 className="text-xl font-bold text-gray-900">Edit Coordinators</h2>
            <p className="text-sm text-gray-500">{session.session_code} - {session.session_name}</p>
          </div>
          <button onClick={onClose} className="text-gray-500 hover:text-gray-800 text-2xl">×</button>
        </div>

        <div className="p-6 space-y-6">
          {/* Current Coordinators List */}
          <div>
            <h3 className="text-sm font-bold text-gray-700 mb-3 uppercase tracking-wider">Current Coordinators</h3>
            {coordinators.length === 0 ? (
              <p className="text-sm text-gray-500 italic bg-gray-50 p-4 rounded-lg border border-gray-200">No coordinators assigned yet.</p>
            ) : (
              <div className="space-y-2">
                {coordinators.map((c, i) => (
                  <div key={i} className="flex items-center justify-between bg-blue-50 border border-blue-200 p-3 rounded-lg">
                    <div>
                      <div className="font-semibold text-blue-900">{c.faculty?.canonical_name}</div>
                      <div className="text-xs font-bold text-blue-700 bg-blue-200 px-2 py-0.5 rounded inline-block mt-1 uppercase">{c.role}</div>
                    </div>
                    <button onClick={() => handleRemoveCoordinator(c.faculty_id || c.faculty.id)} className="text-sm text-red-600 hover:underline font-medium">
                      Remove
                    </button>
                  </div>
                ))}
              </div>
            )}
          </div>

          <hr className="border-gray-200" />

          {/* Add New Coordinator */}
          <div>
            <h3 className="text-sm font-bold text-gray-700 mb-3 uppercase tracking-wider">Add Coordinator</h3>
            
            <div className="grid grid-cols-3 gap-4 mb-4">
              <div className="col-span-1">
                <label className="block text-xs font-bold text-gray-600 mb-1">Role</label>
                <select 
                  value={selectedRole} 
                  onChange={(e) => setSelectedRole(e.target.value)} 
                  className="w-full border border-gray-300 rounded-lg px-3 py-2 bg-white focus:ring-2 focus:ring-blue-500 focus:outline-none text-sm"
                >
                  <option value="National">National</option>
                  <option value="Lead Coordinator">Lead Coordinator</option>
                  <option value="Local">Local</option>
                </select>
              </div>
              <div className="col-span-2 relative">
                <label className="block text-xs font-bold text-gray-600 mb-1">Search Faculty</label>
                <input 
                  type="text" 
                  placeholder="Type name or email..." 
                  className="w-full border border-gray-300 rounded-lg px-3 py-2 focus:ring-2 focus:ring-blue-500 focus:outline-none text-sm"
                  value={searchFaculty}
                  onChange={(e) => setSearchFaculty(e.target.value)}
                />
                
                {/* Search Results Dropdown */}
                {searchFaculty.length > 0 && (
                  <div className="absolute z-10 w-full bg-white border border-gray-200 rounded-lg shadow-lg max-h-48 overflow-y-auto mt-1">
                    {filteredFaculty.length === 0 ? (
                      <div className="p-3 text-sm text-gray-500">No faculty found</div>
                    ) : (
                      filteredFaculty.map(f => (
                        <button
                          key={f.id}
                          onClick={() => handleAddCoordinator(f)}
                          className="w-full text-left p-3 border-b border-gray-100 hover:bg-blue-50 text-sm transition-colors"
                        >
                          <div className="font-semibold text-gray-900">{f.canonical_name}</div>
                          <div className="text-xs text-gray-500">{f.email}</div>
                        </button>
                      ))
                    )}
                  </div>
                )}
              </div>
            </div>
          </div>
        </div>

        <div className="p-5 border-t bg-gray-50 flex justify-end gap-3">
          <button onClick={onClose} className="px-5 py-2.5 border border-gray-300 rounded-lg font-medium hover:bg-gray-100 bg-white">Cancel</button>
          <button onClick={handleSave} disabled={isSaving} className="px-5 py-2.5 bg-blue-600 text-white rounded-lg font-medium hover:bg-blue-700 disabled:opacity-50">
            {isSaving ? 'Saving...' : 'Save Coordinators'}
          </button>
        </div>
      </div>
    </div>
  );
}


===== app\components\EditFacultyModal.tsx =====
'use client';

import { useEffect, useState, useTransition } from 'react';
import { updateFaculty } from '../actions';

type Faculty = {
  id: string;
  canonical_name: string;
  first_name: string | null;
  last_name: string | null;
  email: string;
  phone: string | null;
  designation: string | null;
  is_active: boolean | null;
  faculty_id: string | null;
  country: string | null;
  alt_email: string | null;
  sessions_as_faculty: number | null;
  person_type: string | null;
  is_faculty: boolean | null;
  is_coordinator: boolean | null;
  google_confirmed: boolean | null;
  google_form_sync_date: string | null;
  google_email: string | null;
  google_phone: string | null;
  email_sent: boolean | null;
  email_sent_date: string | null;
  reminder_sent: boolean | null;
  whatsapp_sent: boolean | null;
};

export default function EditFacultyModal({
  faculty,
  onClose,
  onSaved,
}: {
  faculty: Faculty;
  onClose: () => void;
  onSaved: () => void;
}) {
  const [isPending, startTransition] = useTransition();
  const [error, setError] = useState<string | null>(null);
  const [showAdvanced, setShowAdvanced] = useState(false);

  useEffect(() => {
    const originalOverflow = document.body.style.overflow;
    document.body.style.overflow = 'hidden';
    return () => {
      document.body.style.overflow = originalOverflow;
    };
  }, []);

  function handleSubmit(formData: FormData) {
    setError(null);
    startTransition(async () => {
      const result = await updateFaculty(formData);
      if (result.success) {
        onSaved();
        onClose();
      } else {
        setError(result.message ?? 'Update failed');
      }
    });
  }

  return (
    <div className="fixed inset-0 bg-black/50 flex items-center justify-center z-50 p-4">
      <div className="bg-white rounded-lg shadow-xl w-full max-w-2xl max-h-[90vh] flex flex-col overflow-hidden">
        <div className="flex justify-between items-center p-5 border-b shrink-0 bg-white z-10">
          <h2 className="text-xl font-semibold">
            Edit Faculty {faculty.canonical_name ? `- ${faculty.canonical_name}` : ''}
          </h2>
          <button
            type="button"
            onClick={onClose}
            className="text-gray-500 hover:text-gray-800 text-2xl leading-none"
            aria-label="Close"
          >
            ×
          </button>
        </div>

        <form action={handleSubmit} className="flex flex-col flex-1 min-h-0">
          <div className="p-5 space-y-4 overflow-y-auto overscroll-contain flex-1">
            <input type="hidden" name="id" value={faculty.id} />

            <div className="grid grid-cols-2 gap-4">
              <div>
                <label className="block text-sm font-medium mb-1">First name</label>
                <input
                  name="first_name"
                  defaultValue={faculty.first_name ?? ''}
                  className="w-full border rounded px-3 py-2"
                />
              </div>
              <div>
                <label className="block text-sm font-medium mb-1">Last name</label>
                <input
                  name="last_name"
                  defaultValue={faculty.last_name ?? ''}
                  className="w-full border rounded px-3 py-2"
                />
              </div>
            </div>

            <div>
              <label className="block text-sm font-medium mb-1">
                Email <span className="text-red-500">*</span>
              </label>
              <input
                name="email"
                type="email"
                required
                defaultValue={faculty.email ?? ''}
                className="w-full border rounded px-3 py-2"
              />
            </div>

            <div className="grid grid-cols-2 gap-4">
              <div>
                <label className="block text-sm font-medium mb-1">Phone</label>
                <input
                  name="phone"
                  defaultValue={faculty.phone ?? ''}
                  className="w-full border rounded px-3 py-2"
                />
              </div>
              <div>
                <label className="block text-sm font-medium mb-1">Country</label>
                <input
                  name="country"
                  defaultValue={faculty.country ?? ''}
                  className="w-full border rounded px-3 py-2"
                />
              </div>
            </div>

            <div>
              <label className="block text-sm font-medium mb-1">Designation</label>
              <input
                name="designation"
                defaultValue={faculty.designation ?? ''}
                className="w-full border rounded px-3 py-2"
              />
            </div>

            <div className="flex flex-wrap gap-6 pt-2">
              <label className="flex items-center gap-2 text-sm">
                <input type="checkbox" name="is_active" defaultChecked={!!faculty.is_active} />
                Active
              </label>
              <label className="flex items-center gap-2 text-sm">
                <input type="checkbox" name="is_faculty" defaultChecked={!!faculty.is_faculty} />
                Is Faculty
              </label>
              <label className="flex items-center gap-2 text-sm">
                <input type="checkbox" name="is_coordinator" defaultChecked={!!faculty.is_coordinator} />
                Is Coordinator
              </label>
            </div>

            <div className="pt-4 border-t">
              <button
                type="button"
                onClick={() => setShowAdvanced((s) => !s)}
                className="text-sm text-blue-600 hover:underline"
              >
                {showAdvanced ? 'Hide advanced fields' : 'Show advanced fields'}
              </button>

              {showAdvanced && (
                <div className="mt-4 space-y-4 bg-gray-50 p-4 rounded">
                  <p className="text-xs text-gray-500">
                    System / sync fields - normally written by server actions. Only edit if you know what you are doing.
                  </p>

                  <div className="grid grid-cols-2 gap-4">
                    <div>
                      <label className="block text-sm font-medium mb-1">Faculty ID</label>
                      <input
                        name="faculty_id"
                        defaultValue={faculty.faculty_id ?? ''}
                        placeholder="e.g. FAC304"
                        className="w-full border rounded px-3 py-2"
                      />
                    </div>
                    <div>
                      <label className="block text-sm font-medium mb-1">Person type</label>
                      <input
                        name="person_type"
                        defaultValue={faculty.person_type ?? ''}
                        className="w-full border rounded px-3 py-2"
                      />
                    </div>
                  </div>

                  <div className="grid grid-cols-2 gap-4">
                    <div>
                      <label className="block text-sm font-medium mb-1">Alt email</label>
                      <input
                        name="alt_email"
                        type="email"
                        defaultValue={faculty.alt_email ?? ''}
                        className="w-full border rounded px-3 py-2"
                      />
                    </div>
                    <div>
                      <label className="block text-sm font-medium mb-1">Google email</label>
                      <input
                        name="google_email"
                        type="email"
                        defaultValue={faculty.google_email ?? ''}
                        className="w-full border rounded px-3 py-2"
                      />
                    </div>
                  </div>

                  <div className="grid grid-cols-2 gap-4">
                    <div>
                      <label className="block text-sm font-medium mb-1">Google phone</label>
                      <input
                        name="google_phone"
                        defaultValue={faculty.google_phone ?? ''}
                        className="w-full border rounded px-3 py-2"
                      />
                    </div>
                    <div>
                      <label className="block text-sm font-medium mb-1">Sessions as faculty</label>
                      <input
                        name="sessions_as_faculty"
                        type="number"
                        min={0}
                        defaultValue={faculty.sessions_as_faculty ?? ''}
                        className="w-full border rounded px-3 py-2"
                      />
                    </div>
                  </div>

                  <div className="flex flex-wrap gap-6">
                    <label className="flex items-center gap-2 text-sm">
                      <input
                        type="checkbox"
                        name="google_confirmed"
                        defaultChecked={!!faculty.google_confirmed}
                      />
                      Google confirmed
                    </label>
                    <label className="flex items-center gap-2 text-sm">
                      <input type="checkbox" name="email_sent" defaultChecked={!!faculty.email_sent} />
                      Email sent
                    </label>
                    <label className="flex items-center gap-2 text-sm">
                      <input
                        type="checkbox"
                        name="reminder_sent"
                        defaultChecked={!!faculty.reminder_sent}
                      />
                      Reminder sent
                    </label>
                    <label className="flex items-center gap-2 text-sm">
                      <input
                        type="checkbox"
                        name="whatsapp_sent"
                        defaultChecked={!!faculty.whatsapp_sent}
                      />
                      WhatsApp sent
                    </label>
                  </div>

                  <div className="grid grid-cols-2 gap-4 text-xs text-gray-500">
                    <div>
                      <div className="font-medium">Google form synced:</div>
                      <div>
                        {faculty.google_form_sync_date
                          ? new Date(faculty.google_form_sync_date).toLocaleString()
                          : '—'}
                      </div>
                    </div>
                    <div>
                      <div className="font-medium">Email sent date:</div>
                      <div>
                        {faculty.email_sent_date
                          ? new Date(faculty.email_sent_date).toLocaleString()
                          : '—'}
                      </div>
                    </div>
                  </div>
                </div>
              )}
            </div>

            {error && (
              <p className="text-red-600 text-sm bg-red-50 border border-red-200 rounded p-2">
                {error}
              </p>
            )}
          </div>

          <div className="flex justify-end gap-2 p-5 border-t bg-gray-50 shrink-0">
            <button
              type="button"
              onClick={onClose}
              disabled={isPending}
              className="px-4 py-2 border rounded bg-white hover:bg-gray-50 disabled:opacity-50"
            >
              Cancel
            </button>
            <button
              type="submit"
              disabled={isPending}
              className="px-4 py-2 bg-blue-600 text-white rounded hover:bg-blue-700 disabled:opacity-50"
            >
              {isPending ? 'Saving...' : 'Save Changes'}
            </button>
          </div>
        </form>
      </div>
    </div>
  );
}
